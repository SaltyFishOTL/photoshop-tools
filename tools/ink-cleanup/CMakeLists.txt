cmake_minimum_required(VERSION 3.20)
project(InkCleanupNative LANGUAGES C CXX RC)
set(CMAKE_CXX_STANDARD 17)
add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:/EHsc>")
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(ink_engine STATIC engine-native.cpp)
target_compile_options(ink_engine PRIVATE /fp:strict)
target_compile_definitions(ink_engine PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
add_executable(engine_cli engine_cli.cpp)
target_link_libraries(engine_cli PRIVATE ink_engine)
add_executable(cache_test cache-test.cpp)
target_link_libraries(cache_test PRIVATE ink_engine)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/filter.cpp")
set(PHOTOSHOP_SDK_ROOT "" CACHE PATH "Root of the separately obtained Adobe Photoshop C++ SDK")
if(NOT EXISTS "${PHOTOSHOP_SDK_ROOT}/pluginsdk/photoshopapi/photoshop/PIFilter.h")
message(FATAL_ERROR "Set PHOTOSHOP_SDK_ROOT to the Adobe Photoshop C++ SDK root.")
endif()
set(SDK "${PHOTOSHOP_SDK_ROOT}/pluginsdk/photoshopapi")
find_program(NODE_EXECUTABLE node REQUIRED)
execute_process(COMMAND "${NODE_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate.js" "${PHOTOSHOP_SDK_ROOT}" COMMAND_ERROR_IS_FATAL ANY)
add_library(InkCleanup SHARED filter.cpp filter.rc)
target_link_libraries(InkCleanup PRIVATE ink_engine comctl32 user32 gdi32)
target_include_directories(InkCleanup PRIVATE "${SDK}/photoshop" "${SDK}/pica_sp" "${SDK}/resources" "${SDK}/../samplecode/common/includes")
target_compile_definitions(InkCleanup PRIVATE WIN_ENV=1 MSWindows=1 _WINDOWS NOMINMAX UNICODE _UNICODE)
target_compile_options(InkCleanup PRIVATE /utf-8)
set_target_properties(InkCleanup PROPERTIES PREFIX "" SUFFIX ".8bf")
add_executable(host_test host-test.cpp)
target_include_directories(host_test PRIVATE "${SDK}/photoshop" "${SDK}/pica_sp" "${SDK}/../samplecode/common/includes")
target_compile_definitions(host_test PRIVATE NOMINMAX)
add_executable(ui_test ui-test.cpp)
target_include_directories(ui_test PRIVATE "${SDK}/photoshop" "${SDK}/pica_sp" "${SDK}/../samplecode/common/includes")
target_compile_definitions(ui_test PRIVATE NOMINMAX)
target_compile_options(ui_test PRIVATE /utf-8)
endif()
